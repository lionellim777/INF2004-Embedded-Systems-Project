cmake_minimum_required(VERSION 3.20)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(terrain_representation_test C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(terrain_representation_test terrain_representation_test.c)
target_include_directories(terrain_representation_test PRIVATE ../../app/robot_firmware)
target_compile_options(terrain_representation_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(terrain_representation_test pico_stdlib hardware_i2c m)
pico_enable_stdio_usb(terrain_representation_test 1)
pico_enable_stdio_uart(terrain_representation_test 0)
pico_add_extra_outputs(terrain_representation_test)
